    <script>
        const urlParams = new URLSearchParams(window.location.search);
        const guestName = urlParams.get('guest');
        if (guestName) {
            document.getElementById('guestWelcome').innerHTML = `<i class="fa-solid fa-heart"></i> أهلاً بك يا ${decodeURIComponent(guestName)}`;
        }

        const music = document.getElementById('bgMusic');
        const musicIcon = document.getElementById('musicIcon');
        let isPlaying = false;

        function toggleMusic() {
            if (isPlaying) {
                music.pause();
                musicIcon.className = "fa-solid fa-music";
                isPlaying = false;
            } else {
                music.play().then(() => {
                    musicIcon.className = "fa-solid fa-volume-high";
                    isPlaying = true;
                }).catch(e => { console.log("Audio play blocked"); });
            }
        }

        const weddingDate = new Date("October 16, 2026 19:00:00").getTime();
        const timer = setInterval(function() {
            const now = new Date().getTime();
            const distance = weddingDate - now;

            if (distance < 0) {
                clearInterval(timer);
                document.getElementById("countdown").innerHTML = "<div style='color:#d4af37; font-weight:bold;'>اليوم المنتظر قد بدأ! مبارك للعرسان</div>";
                return;
            }

            document.getElementById("days").innerText = Math.floor(distance / (1000 * 60 * 60 * 24));
            document.getElementById("hours").innerText = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
            document.getElementById("minutes").innerText = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
            document.getElementById("seconds").innerText = Math.floor((distance % (1000 * 60)) / 1000);
        }, 1000);
    </script>
